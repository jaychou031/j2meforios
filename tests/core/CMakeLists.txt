cmake_minimum_required(VERSION 3.24)
project(J2MECoreChecks C CXX)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(CORE "${ROOT}/upstream/esp32s31-korvo-nano-j2me/third_party/nano_j2me")
foreach(option NJ_BUILD_TESTS NJ_BUILD_SDL2 NJ_BUILD_LIBRETRO NJ_BUILD_EMBEDDED_PORTS NJ_BUILD_POSIX_NETWORK)
  set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
set(NJ_BUILD_CLI ON CACHE BOOL "" FORCE)
set(NJ_ENABLE_MEDIA 1 CACHE STRING "" FORCE)
set(NJ_ENABLE_JIT 0 CACHE STRING "" FORCE)
set(NJ_PIXFMT NJ_PIXFMT_ARGB8888 CACHE STRING "" FORCE)
add_subdirectory("${CORE}" core)
target_compile_definitions(nj2me PUBLIC NJ_USE_LIBC=1)
if(APPLE)
  target_compile_definitions(nj2me PRIVATE _DARWIN_C_SOURCE=1)
endif()
include("${ROOT}/cmake/CoreCompatibility.cmake")
j2me_apply_core_compatibility(nj2me "${CORE}")
add_executable(port-stats-test "${ROOT}/tests/port_stats_test.c")
target_link_libraries(port-stats-test PRIVATE nj2me)
add_executable(soundfont-test "${ROOT}/tests/soundfont_test.cpp")
target_include_directories(soundfont-test PRIVATE "${CORE}/src/midp" "${ROOT}/ios")
target_link_libraries(soundfont-test PRIVATE nj2me)
add_executable(render-midi "${ROOT}/tests/render_midi.cpp")
target_include_directories(render-midi PRIVATE "${CORE}/src/midp" "${ROOT}/ios")
target_link_libraries(render-midi PRIVATE nj2me)
add_executable(soundfont-runner "${ROOT}/tests/soundfont_runner.cpp")
target_include_directories(soundfont-runner PRIVATE "${ROOT}/ios")
target_link_libraries(soundfont-runner PRIVATE nj2me)
add_executable(idle-pacer-test "${ROOT}/tests/idle_pacer_test.c")
target_include_directories(idle-pacer-test PRIVATE "${CORE}/src/core" "${ROOT}/ios")
target_link_libraries(idle-pacer-test PRIVATE nj2me)
